## 3. Filtering, calculations, and assistant logic

### `src/lib/analysis.ts`

```ts
import type {
  ComparisonResult,
  Experiment,
  Filters,
} from "../types";

export const defaultFilters: Filters = {
  query: "",
  type: "",
  fuel: "",
  oxygen: "",
  gravity: "",
  availability: "",
  sort: "name",
};

export function hasTemperature(experiment: Experiment): boolean {
  return experiment.measurements.some((point) => point.temperatureK !== null);
}

export function hasOptical(experiment: Experiment): boolean {
  return experiment.measurements.some((point) => point.luminosityAU !== null);
}

export function oxygenBand(oxygen: number): string {
  if (oxygen < 21) return "Below 21%";
  if (oxygen === 21) return "21%";
  return "Above 21%";
}

export function filterExperiments(
  records: Experiment[],
  filters: Filters,
): Experiment[] {
  const query = filters.query.trim().toLowerCase();

  const filtered = records.filter((experiment) => {
    const searchable = [
      experiment.id,
      experiment.name,
      experiment.type,
      experiment.fuel,
      experiment.gravity,
      experiment.description,
      experiment.flameShape,
    ]
      .join(" ")
      .toLowerCase();

    const availabilityMatches =
      !filters.availability ||
      (filters.availability === "temperature" && hasTemperature(experiment)) ||
      (filters.availability === "optical" && hasOptical(experiment)) ||
      (filters.availability === "both" &&
        hasTemperature(experiment) &&
        hasOptical(experiment));

    return (
      (!query || searchable.includes(query)) &&
      (!filters.type || experiment.type === filters.type) &&
      (!filters.fuel || experiment.fuel === filters.fuel) &&
      (!filters.gravity || experiment.gravity === filters.gravity) &&
      (!filters.oxygen || oxygenBand(experiment.oxygenPct) === filters.oxygen) &&
      availabilityMatches
    );
  });

  return filtered.sort((a, b) => {
    switch (filters.sort) {
      case "date":
        return b.scenarioDate.localeCompare(a.scenarioDate);
      case "duration":
        return b.durationS - a.durationS;
      case "temperature":
        return (b.peakTemperatureK ?? -Infinity) -
          (a.peakTemperatureK ?? -Infinity);
      default:
        return a.name.localeCompare(b.name);
    }
  });
}

export function mean(values: number[]): number | null {
  return values.length
    ? values.reduce((sum, value) => sum + value, 0) / values.length
    : null;
}

export function format(value: number | null, digits = 1): string {
  return value === null
    ? "Unavailable"
    : new Intl.NumberFormat("en-US", {
        maximumFractionDigits: digits,
      }).format(value);
}

export interface Aggregate {
  name: string;
  value: number;
  count: number;
}

export function groupDuration(
  records: Experiment[],
  key: "fuel" | "gravity" | "oxygenPct",
): Aggregate[] {
  const groups = new Map<string, number[]>();

  for (const experiment of records) {
    const label =
      key === "oxygenPct"
        ? `${experiment.oxygenPct}%`
        : String(experiment[key]);
    groups.set(label, [...(groups.get(label) ?? []), experiment.durationS]);
  }

  return [...groups.entries()]
    .map(([name, values]) => ({
      name,
      value: Number((mean(values) ?? 0).toFixed(1)),
      count: values.length,
    }))
    .sort((a, b) =>
      a.name.localeCompare(b.name, undefined, { numeric: true }),
    );
}

export function durationDistribution(records: Experiment[]): Aggregate[] {
  return [
    { low: 0, high: 40 },
    { low: 40, high: 80 },
    { low: 80, high: 120 },
    { low: 120, high: 160 },
  ].map(({ low, high }) => {
    const count = records.filter(
      (experiment) =>
        experiment.durationS >= low && experiment.durationS < high,
    ).length;
    return { name: `${low}–<${high}`, value: count, count };
  });
}

export function compareExperiments(
  records: Experiment[],
): ComparisonResult {
  if (records.length < 2) {
    return {
      experimentIds: records.map((record) => record.id),
      durationRangeS: 0,
      oxygenRangePct: 0,
      temperatureRecordCount: records.filter(hasTemperature).length,
      differingFields: [],
      summary: "Select at least two fictional scenarios to compare.",
    };
  }

  const durationValues = records.map((record) => record.durationS);
  const oxygenValues = records.map((record) => record.oxygenPct);
  const durationRangeS = Math.max(...durationValues) - Math.min(...durationValues);
  const oxygenRangePct = Math.max(...oxygenValues) - Math.min(...oxygenValues);

  const fields: Array<[keyof Experiment, string]> = [
    ["fuel", "Fuel"],
    ["oxygenPct", "Oxygen"],
    ["gravity", "Gravity"],
    ["flameShape", "Flame shape"],
    ["durationS", "Duration"],
    ["peakTemperatureK", "Peak temperature"],
    ["flowCms", "Flow"],
  ];

  const differingFields = fields
    .filter(([key]) => new Set(records.map((record) => record[key])).size > 1)
    .map(([, label]) => label);

  const temperatureRecordCount = records.filter(hasTemperature).length;

  return {
    experimentIds: records.map((record) => record.id),
    durationRangeS,
    oxygenRangePct,
    temperatureRecordCount,
    differingFields,
    summary:
      `Across ${records.length} selected fictional scenarios, duration spans ` +
      `${Math.min(...durationValues)}–${Math.max(...durationValues)} s ` +
      `(range ${durationRangeS} s), while oxygen spans ` +
      `${Math.min(...oxygenValues)}–${Math.max(...oxygenValues)}%. ` +
      `${temperatureRecordCount} of ${records.length} records include temperature series. ` +
      "These descriptive differences do not establish causation; fuel, flow, " +
      "gravity, and other conditions may differ. No uncertainty estimates or " +
      "experimental replicates are available.",
  };
}

export function datasetInsights(records: Experiment[]): string[] {
  if (!records.length) return [];

  const average = mean(records.map((record) => record.durationS));
  const longest = records.reduce((a, b) => a.durationS > b.durationS ? a : b);
  const temperatureCount = records.filter(hasTemperature).length;

  return [
    `Mean fictional combustion duration is ${format(average)} s across ${records.length} selected scenarios.`,
    `${longest.id} has the longest duration in this selection: ${longest.durationS} s. This is a catalog ranking, not an experimental conclusion.`,
    `${temperatureCount} of ${records.length} selected scenarios include synthetic temperature data; missing values are excluded rather than imputed.`,
  ];
}
```

```ts
import type { AssistantReply } from "../types";

export const exampleQuestions = [

  "Compare combustion behavior under different oxygen concentrations.",
  "Which experiments have temperature measurements?",
  "What are the limitations of these experiments?",
];

function makeReply(
  text: string,
  ids: string[],
  followUps = exampleQuestions.slice(1, 3),
): AssistantReply {
  return {
    text,
    experimentIds: ids,
    sourceIds: ["demo-catalog"],
    followUps,
  };
}

/**
 * Local, deliberately limited keyword dispatcher.
 * Replace this adapter with an API client later; preserve the evidence schema.
 * User text is never executed and responses are rendered as plain React text.
 */
export function answerQuestion(
  question: string,
  selectedIds: string[],
): AssistantReply {
  const query = question.toLowerCase();

  if (/limitation|uncertaint|reliab|real data|source|nasa/.test(query)) {
    return makeReply(
      "All eight records are fictional UI demonstration scenarios. Their curves " +
      "are deterministic illustrations, not measurements or validated simulations. " +
      "There are no experimental replicates, calibration records, uncertainty " +
      "estimates, or verified experiment-level NASA citations. Some records have " +
      "no temperature series. General NASA links are background references only. " +
      "This catalog cannot support operational fire-safety decisions or causal claims.",
      ["FF-05", "FF-08"],
      [exampleQuestions[2], "Compare my selected experiments."],
    );
  }

  if (/temperature|thermal|kelvin/.test(query)) {
    const records = experiments.filter(hasTemperature);
    return makeReply(
      `${records.length} fictional scenarios include temperature series: ` +
      records
        .map((record) => `${record.id} (${record.name}, peak ${record.peakTemperatureK} K)`)
        .join("; ") +
      ". FF-05 and FF-08 have no temperature measurements. These synthetic peak " +
      "values are catalog metadata, not verified NASA findings.",
      records.map((record) => record.id),
      [exampleQuestions[3], "Compare my selected experiments."],
    );
  }

  if (/oxygen|concentration/.test(query)) {
    const groups = groupDuration(experiments, "oxygenPct");
    return makeReply(
      "Grouped by oxygen concentration, mean fictional duration is: " +
      groups.map(
        (group) => `${group.name}: ${format(group.value)} s (n=${group.count})`,
      ).join("; ") +
      ". The groups also differ in fuel, gravity, and flow. These small, authored " +
      "groups do not isolate an oxygen effect and cannot establish a causal trend.",
      ["FF-03", "FF-04", "FF-01"],
      [exampleQuestions[3], "Compare my selected experiments."],
    );
  }

  if (/shape|microgravity|gravity/.test(query)) {
    const a = experiments[0];
    const b = experiments[1];
    return makeReply(
      `The demo labels ${a.id} (${a.gravity}) as “${a.flameShape}” and ` +
      `${b.id} (${b.gravity}) as “${b.flameShape}.” Both use methanol at 21% ` +
      "oxygen in this authored catalog. In combustion research, reduced buoyancy " +
      "is a relevant physical consideration, but these fictional descriptors do " +
      "not demonstrate that mechanism or imply that every microgravity flame is " +
      "spherical. Validate physical explanations against experiment-level sources " +
      "before using them as research evidence.",
      [a.id, b.id],
      [exampleQuestions[1], exampleQuestions[3]],
    );
  }

  if (/compare|selected|difference/.test(query)) {
    const mentioned = experiments.filter((record) =>
      query.includes(record.id.toLowerCase()) ||
      query.includes(record.name.toLowerCase()),
    );
    const selected = experiments.filter((record) => selectedIds.includes(record.id));
    const records = mentioned.length >= 2 ? mentioned : selected;

    return makeReply(
      compareExperiments(records).summary,
      records.map((record) => record.id),
      [exampleQuestions[1], exampleQuestions[3]],
    );
  }

  const record = experiments.find((experiment) =>
    query.includes(experiment.id.toLowerCase()) ||
    query.includes(experiment.name.toLowerCase()),
  );

  if (record) {
    return makeReply(
      `${record.id} — ${record.name}: fictional ${record.fuel} scenario, ` +
      `${record.oxygenPct}% oxygen, ${record.gravity}, ${record.durationS} s duration. ` +
```

```tsx
const initialMessage: ChatMessage = {

  text:

interface WorkspaceValue {
  page: Page;
  navigate: (page: Page) => void;
  query: string;
  setQuery: (query: string) => void;
  selectedIds: string[];
  toggleSelection: (id: string) => void;
  clearSelection: () => void;
  detailId: string | null;
  openDetail: (id: string) => void;
  closeDetail: () => void;
  messages: ChatMessage[];
  addMessage: (message: ChatMessage) => void;
  resetMessages: () => void;
}

const WorkspaceContext = createContext<WorkspaceValue | null>(null);

export function WorkspaceProvider({ children }: { children: ReactNode }) {
  const [page, setPage] = useState<Page>(readPage);
  const [query, setQuery] = useState("");
  const [selectedIds, setSelectedIds] = useState(["FF-01", "FF-02"]);
  const [detailId, setDetailId] = useState<string | null>(null);
  const [messages, setMessages] = useState<ChatMessage[]>([initialMessage]);

  useEffect(() => {
    const listener = () => setPage(readPage());
    window.addEventListener("hashchange", listener);
    return () => window.removeEventListener("hashchange", listener);
  }, []);

  function navigate(next: Page) {
    setPage(next);
    window.location.hash = next;
    window.scrollTo({ top: 0 });
  }

  function toggleSelection(id: string) {
    if (!experiments.some((record) => record.id === id)) return;
    setSelectedIds((previous) =>
      previous.includes(id)
        ? previous.filter((selected) => selected !== id)
        : [...previous, id],
    );
  }

  return (
    <WorkspaceContext.Provider
      value={{
        page,
        navigate,
        query,
        setQuery,
        selectedIds,
        toggleSelection,
        clearSelection: () => setSelectedIds([]),
        detailId,
        openDetail: setDetailId,
        closeDetail: () => setDetailId(null),
        messages,
        addMessage: (message) => setMessages((previous) => [...previous, message]),
        resetMessages: () => setMessages([initialMessage]),
      }}
    >
      {children}
    </WorkspaceContext.Provider>
  );
}

export function useWorkspace(): WorkspaceValue {
  const context = useContext(WorkspaceContext);
  if (!context) throw new Error("WorkspaceProvider is missing.");
  return context;
```

```tsx
export function TimeSeries({
  records,
  metric = "temperatureK",
}: {
  records: Experiment[];
  metric?: "temperatureK" | "luminosityAU";
}) {
  const available = records.filter((record) =>
    record.measurements.some((point) => point[metric] !== null),
  );

  if (!available.length) {
    return (
      <EmptyState
        title="This measurement is unavailable"
        description="Missing series are not filled with estimated values."
      />
    );
  }

  const rows = new Map<number, Record<string, number | null>>();

  for (const record of available) {
    for (const point of record.measurements) {
      const row = rows.get(point.timeS) ?? { timeS: point.timeS };
      row[record.id] = point[metric];
      rows.set(point.timeS, row);
    }
  }

  const data = [...rows.values()].sort(
    (a, b) => (a.timeS ?? 0) - (b.timeS ?? 0),
  );
  const unit = metric === "temperatureK" ? "K" : "a.u.";
  const label = metric === "temperatureK" ? "Temperature" : "Luminosity";

  return (
    <div
      className="h-80 w-full min-w-0"
      role="img"

  );
```

```tsx
    const previousFocus = document.activeElement as HTMLElement | null;
    const previousOverflow = document.body.style.overflow;
    document.body.style.overflow = "hidden";
    closeButton.current?.focus();

    function handleKey(event: KeyboardEvent) {
      if (event.key === "Escape") onClose();

      if (event.key === "Tab") {

          event.preventDefault();
          last.focus();
        } else if (!event.shiftKey && document.activeElement === last) {
          event.preventDefault();
          first.focus();
        }
      }
    }

    document.addEventListener("keydown", handleKey);
    return () => {
      document.body.style.overflow = previousOverflow;
      document.removeEventListener("keydown", handleKey);
```

```tsx
import { useEffect, useRef, useState, type FormEvent } from "react";

export default function Copilot() {
  const {
    selectedIds,
    messages,
    addMessage,
    resetMessages,
    openDetail,
    navigate,
  } = useWorkspace();

  const [input, setInput] = useState("");
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const pending = useRef(false);
  const bottom = useRef<HTMLDivElement>(null);
  const mounted = useRef(true);

  useEffect(() => {
    mounted.current = true;
    return () => { mounted.current = false; };
  }, []);

  useEffect(() => {
    const scroller = bottom.current?.parentElement;
    if (scroller) scroller.scrollTop = scroller.scrollHeight;
  }, [messages.length, busy]);

  async function send(question: string) {
    const text = question.trim();
    if (!text || pending.current) return;

    pending.current = true;
    setBusy(true);
    setError("");
    setInput("");

    addMessage({ id: crypto.randomUUID(), role: "user", text });

    try {
      const response = await askDemo(text, selectedIds);
      // Keep the reply in shared history even if the user changed pages.
      addMessage({
        id: crypto.randomUUID(),
        role: "assistant",
        text: response.text,
        evidence: response,
      });
    } catch {
      if (mounted.current) {
        setError("The local demo could not generate a reply. Your question is retained; try sending it again.");
        setInput(text);
      }
    } finally {
      pending.current = false;
      if (mounted.current) setBusy(false);
    }
  }

  function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    void send(input);
  }

  const latest = [...messages].reverse().find((message) => message.evidence);
  const followUps = latest?.evidence?.followUps ?? exampleQuestions.slice(0, 2);

  return (
    <>
      <PageIntro
        eyebrow="RESEARCH COPILOT / LOCAL DEMO"

            <div className="flex items-center gap-3">
              <span className="rounded-lg border border-cyan-300/20 bg-cyan-300/5 p-2 text-cyan-200"><Bot size={20} /></span>
              <div>
                <h2 className="text-sm font-semibold">FIREFLY Research Copilot</h2>
                <p className="mt-1 text-xs text-muted">Local rules · {selectedIds.length} comparison records selected</p>
              </div>
            </div>
            <button className="text-button" disabled={busy} onClick={resetMessages}>Clear chat</button>
          </div>

          <div
            className="h-[520px] space-y-6 overflow-y-auto p-4 sm:p-6"
            role="log"
            aria-label="Research conversation"
            aria-live="polite"
          >
            {messages.map((message) => (
              <article key={message.id} className="flex items-start gap-3">
                <div className={`mt-1 rounded-lg border p-2 ${message.role === "assistant" ? "border-cyan-300/20 text-cyan-200" : "border-line text-muted"}`}>

                      <div className="mt-4 border-t border-line pt-4">

                              <button key={id} onClick={() => openDetail(id)} className="evidence-card">
                                <span>

                                  <span className="mt-1 block text-xs">{record.name}</span>
                                </span>
                                <ArrowUpRight size={15} className="shrink-0 text-muted" />

                          })}
```

```md
## Data conventions

- Temperature: kelvin (K)
- Duration/time: seconds (s)
- Oxygen: percent by volume
- Pressure: kilopascals (kPa)
- Flow: centimeters per second (cm/s)
- Gravity: multiples of standard Earth gravity (g)
- Luminosity: arbitrary units (a.u.)
- Missing measurements: null, not zero
- Scenario dates: fictional metadata, not retrieval or access dates

Synthetic temperatures begin at 295 K and follow a simple sine envelope.
Optical curves use a synthetic normalized envelope.
Neither curve generator is a validated physical model.

Missing temperature series are excluded rather than imputed.
```
